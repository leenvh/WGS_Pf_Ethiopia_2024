genes <- c("CRT", "PX1","Kelch13")

gene_results <- list()

for (gene in genes) {

  gene_results[[gene]] <- seekdeep_result |>
    filter(Gene == gene) |>
    # arrange is_reference so reference allele is first when arranging
    arrange(desc(is_reference)) |>
    group_by(Sample, reference_AA_pos) |>
    summarise(
      # take the first reference_AA to avoid duplicate rows for overlapping references (e.g. position 1677 in PX1)
      reference_AA = first(reference_AA),
      # collapse unique AA alleles seen for this positioning
      AA = paste0(unique(AA), collapse = "/"),
      .groups = "drop"
    ) |>
    mutate(
      aa_id = paste0(
        reference_AA,
        reference_AA_pos,
        ifelse(reference_AA != AA, AA, "")
      ),
      gene_pos = paste0(
        gene, "_",
        stringr::str_pad(reference_AA_pos, pad = "0", width = 3)
      )
    ) |>
    select(Sample, gene_pos, aa_id) |>
    tidyr::pivot_wider(
      names_from = gene_pos,
      values_from = aa_id
    )
}

  gene_results[[gene]] <- seekdeep_result |>
    filter(Gene == gene) |>
    # arrange is_reference so reference allele is first when arranging
    arrange(desc(is_reference)) |>
    group_by(Sample, reference_AA_pos) |>
    summarise(
      # take the first reference_AA to avoid duplicate rows for overlapping references (e.g. position 1677 in PX1)
      reference_AA = first(reference_AA),
      # collapse unique AA alleles seen for this positioning
      AA = paste0(unique(AA), collapse = "/"),
      .groups = "drop"
    ) |>
    mutate(
      aa_id = paste0(
        reference_AA,
        reference_AA_pos,
        ifelse(reference_AA != AA, AA, "")
      ),
      gene_pos = paste0(
        gene, "_",
        stringr::str_pad(reference_AA_pos, pad = "0", width = 3)
      )
    ) |>
    select(Sample, gene_pos, aa_id) |>
    tidyr::pivot_wider(
